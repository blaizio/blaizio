}

}

}

  animation: faq-up 0.2s ease-out forwards;
}

@keyframes faq-down {
  from { height: 0; }
  to { height: var(--bz-collapsible-content-height); }
}

  from { height: var(--bz-collapsible-content-height); }
  to { height: 0; }
