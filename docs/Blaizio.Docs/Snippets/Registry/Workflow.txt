name: registry

on:
  push:
    branches: [main]

jobs:
  publish:
    runs-on: ubuntu-latest
    permissions:
      contents: read
      pages: write
      id-token: write
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-dotnet@v4
        with:
          dotnet-version: '10.0.x'
      - run: dotnet tool install -g Blaizio.Cli
      - run: blaizio registry validate ./registry.json
      - run: blaizio build ./registry.json -o ./wwwroot/r
      - uses: actions/upload-pages-artifact@v3
        with:
          path: ./wwwroot
      - uses: actions/deploy-pages@v4
