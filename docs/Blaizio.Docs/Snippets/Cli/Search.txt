# List the configured registry's items:
blaizio search

# -q filters name/title/description; -l and -o, --offset page (default limit 100):
blaizio search -q input,date
blaizio search -l 20 -o 20

# -t filters by item type, --category by the registry's tags:
blaizio search -t theme,font
blaizio search --category forms,data

# Positional args are extra registries - @namespaces, URLs or local paths:
blaizio search @acme https://example.com/r
