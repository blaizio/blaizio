# Re-style an existing project from a preset name:
blaizio apply eclipse

# Or from a Create code (style + color + chart + radius + fonts):
blaizio apply 32r
blaizio apply 000h60

# Apply only parts of the preset:
blaizio apply 000h60 --only theme
blaizio apply 000h60 --only fonts      # also overrides your own font setup
blaizio apply 000h60 --only tokens

# Same thing folded into an add:
blaizio add button -p 000h60
