# Interactive - prompts for template, namespace, output, skin, RTL, Tailwind pipeline:
blaizio init

# Everything up front, no prompts:
blaizio init -t showcase -n MyApp --style aura --tailwind standalone

# Existing app, custom namespace + output, and add components in the same run:
blaizio init --namespace MyApp.Ui -o Components/Ui button card dialog

# Accept every default (template showcase, skin ember, pipeline auto):
blaizio init -d
