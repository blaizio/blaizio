# Existing app - interactive: prompts for namespace, output, skin, preset, RTL:
blaizio init

# Everything up front, no prompts:
blaizio init --namespace MyApp.Ui -o Components/Ui --style aura --tailwind standalone

# Wire up and add components in the same run:
blaizio init button card dialog

# Accept every default (skin ember, preset nova, pipeline auto):
blaizio init -d

# Starting from nothing? `new` scaffolds a whole app, then runs init for you:
blaizio new showcase -n MyApp
