# One component - its registry dependencies (and NuGet packages) come along.
# A project without blaizio.json is wired first (config, packages, tokens file, host page):
blaizio add button

# Several at once, or the whole registry:
blaizio add dialog toast tooltip
blaizio add --all

# Interactive: no arguments opens the component picker (prompts wire a fresh project too):
blaizio add

# See what WOULD be written without touching anything:
blaizio add dialog --dry-run

# Re-copy existing files. Anything you changed since install is listed for you to pick;
# -y keeps your version of those, --force-overwrite replaces them without asking:
blaizio add dialog --overwrite
blaizio add dialog --overwrite -y
blaizio add dialog --overwrite --force-overwrite

# Skip the NuGet install (monorepo, ProjectReference):
blaizio add dialog --no-nuget

# Apply a preset (name or Create code) in the same run - no separate `apply` needed:
blaizio add button -p comet
blaizio add card dialog -p 32r

# Every wiring flag up front - a fresh project needs only one command:
blaizio add button --namespace MyApp.Ui -o Components/Ui --style aura --tailwind standalone
blaizio add --rtl                # wiring-only top-up on an initialized project
blaizio add button -d            # accept every default (skin ember, preset nova, pipeline auto)

# Fonts are registry items too - body and heading halves merge:
blaizio add font-inter
blaizio add font-heading-space-grotesk

# Pin a version the registry stamps - recorded, honored by update/--diff, and
# respected when other items depend on it. A plain re-add unpins:
blaizio add button@1.2.0
blaizio add button

# Items don't have to come from the configured registry:
blaizio add ./my-item.json
blaizio add https://example.com/r/fancy-button.json

# Inspect instead of writing: the upstream diff, or a file's contents:
blaizio add --diff               # exit 1 on drift, like git diff --exit-code
blaizio add button --diff
blaizio add button --view Components/Ui/Button/BzButton.razor
