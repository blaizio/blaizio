# One component - its registry dependencies (and NuGet packages) come along:
blaizio add button

# Several at once, or the whole registry:
blaizio add dialog toast tooltip
blaizio add --all

# See what WOULD be written without touching anything:
blaizio add dialog --dry-run

# Re-copy over local edits / skip the NuGet install (monorepo, ProjectReference):
blaizio add dialog --overwrite
blaizio add dialog --no-nuget

# Items don't have to come from the configured registry:
blaizio add ./my-item.json
blaizio add https://example.com/r/fancy-button.json
