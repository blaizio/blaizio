# What's installed, and has anything drifted from the registry? (exit 1 on drift)
blaizio info
blaizio diff
blaizio diff button

# Browse the registry:
blaizio list -q input
blaizio search date
blaizio view dialog          # metadata + full file contents, written nowhere

# Sync: update re-pulls source (overwrites local edits); upgrade also bumps the
# Blaizio.Base / Blaizio.Icons NuGet packages to this tool's pinned versions first.
blaizio update
blaizio upgrade

# Undo init: remove the Blaizio config, managed CSS and Tailwind wiring.
# Copied components, usings and packages stay untouched.
blaizio deinit --dry-run     # preview first
blaizio deinit
