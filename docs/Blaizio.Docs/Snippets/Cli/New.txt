# Interactive - prompts for the template, then namespace, skin, preset, RTL:
blaizio new

# The full demo app, everything up front:
blaizio new showcase -n MyApp --style aura --tailwind standalone

# A component class library:
blaizio new library -n My.Components

# Accept every default (template showcase, skin ember, pipeline auto):
blaizio new -d

# A registry-hosted template (a registry:template item): scaffolds its files with the
# {{ProjectName}}/{{RootNamespace}}/{{ComponentNamespace}}/{{Skin}} tokens substituted,
# wires the project, then installs the template's component set:
blaizio new starter-saas --registry https://acme.dev/r -n MyApp
blaizio new @acme/starter-saas -n MyApp
